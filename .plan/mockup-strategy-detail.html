from AlgorithmImports import *

class RSIPullbackAlgorithm(QCAlgorithm):

    def Initialize(self):
        self.SetStartDate(2022, 1, 1)
        self.SetEndDate(2024, 1, 1)
        self.SetCash(100000)

        # Add SPY with daily resolution
        self.spy = self.AddEquity("SPY", Resolution.Daily).Symbol
        self.aapl = self.AddEquity("AAPL", Resolution.Daily).Symbol

        # RSI indicators
        self.rsi_spy = self.RSI(self.spy, 14, MovingAverageType.Simple, Resolution.Daily)
        self.rsi_aapl = self.RSI(self.aapl, 14, MovingAverageType.Simple, Resolution.Daily)

    def OnData(self, data):
        if not (self.rsi_spy.IsReady and self.rsi_aapl.IsReady):
            return

        # Buy on RSI oversold (< 30)
        if self.rsi_spy.Current.Value < 30 and not self.Portfolio[self.spy].Invested:
            self.SetHoldings(self.spy, 0.5)

        # Sell on RSI overbought (> 70)
        elif self.rsi_spy.Current.Value > 70 and self.Portfolio[self.spy].Invested:
            self.Liquidate(self.spy)
